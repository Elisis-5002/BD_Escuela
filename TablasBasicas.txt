CREATE TABLE Profesores (
    id_profesor NUMBER PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    apellido VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL
);

CREATE TABLE Cursos (
    id_curso NUMBER PRIMARY KEY,
    nombre_curso VARCHAR(100) NOT NULL,
    id_profesor INT,
    FOREIGN KEY (id_profesor) REFERENCES Profesores(id_profesor)
);

CREATE TABLE Alumnos (
    id_alumno NUMBER PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    apellido VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL
);

CREATE TABLE Inscripciones (
    id_inscripcion NUMBER PRIMARY KEY,
    id_alumno NUMBER NOT NULL,
    id_curso NUMBER NOT NULL,
    FOREIGN KEY (id_alumno) REFERENCES Alumnos(id_alumno),
    FOREIGN KEY (id_curso) REFERENCES Cursos(id_curso)
);

CREATE TABLE Calificaciones (
    id_calificacion NUMBER PRIMARY KEY,
    id_inscripcion NUMBER NOT NULL,
    descripcion VARCHAR(100),
    nota DECIMAL(4,2) NOT NULL,
    fecha DATE NOT NULL,
    FOREIGN KEY (id_inscripcion) REFERENCES Inscripciones(id_inscripcion)
);

CREATE TABLE Asistencia (
    id_asistencia NUMBER PRIMARY KEY,
    id_inscripcion NUMBER NOT NULL,
    fecha DATE NOT NULL,
    estado VARCHAR(20) NOT NULL,
    FOREIGN KEY (id_inscripcion) REFERENCES Inscripciones(id_inscripcion)
);
