-- =====================================================
-- TABLA: USUARIOS
-- =====================================================

CREATE TABLE usuarios (
    usr_id              NUMBER(10)
        GENERATED BY DEFAULT AS IDENTITY,

    usr_nombre          VARCHAR2(16) NOT NULL,

    usr_pwd             CHAR(32) NOT NULL,

    usr_creado          DATE DEFAULT SYSDATE,

    ses_id              NUMBER(10),

    CONSTRAINT pk_usuarios
        PRIMARY KEY (usr_id),

    CONSTRAINT uk_usuarios_nombre
        UNIQUE (usr_nombre)
);



-- =====================================================
-- TABLA: SESIONES
-- =====================================================

CREATE TABLE sesiones (
    ses_id                  NUMBER(10)
        GENERATED BY DEFAULT AS IDENTITY,

    ses_token               CHAR(32) NOT NULL,

    ses_inicio              DATE DEFAULT SYSDATE,

    ses_ultima_actividad    DATE,

    ses_fin                 DATE,

    ses_ip                  VARCHAR2(15),

    usr_id                  NUMBER(10) NOT NULL,

    CONSTRAINT pk_sesiones
        PRIMARY KEY (ses_id),

    CONSTRAINT uk_sesiones_token
        UNIQUE (ses_token),

    CONSTRAINT fk_sesiones_usuarios
        FOREIGN KEY (usr_id)
        REFERENCES usuarios (usr_id)
);



-- =====================================================
-- TABLA: ROLES
-- =====================================================

CREATE TABLE roles (
    rol_id              NUMBER(10)
        GENERATED BY DEFAULT AS IDENTITY,

    rol_nombre          VARCHAR2(32) NOT NULL,

    rol_creado          DATE DEFAULT SYSDATE,

    ses_id              NUMBER(10),

    CONSTRAINT pk_roles
        PRIMARY KEY (rol_id),

    CONSTRAINT uk_roles_nombre
        UNIQUE (rol_nombre),

    CONSTRAINT fk_roles_sesiones
        FOREIGN KEY (ses_id)
        REFERENCES sesiones (ses_id)
);



-- =====================================================
-- TABLA: PERMISOS
-- =====================================================

CREATE TABLE permisos (
    perm_id             NUMBER(10)
        GENERATED BY DEFAULT AS IDENTITY,

    perm_permisos       CHAR(32) NOT NULL,

    perm_creado         DATE DEFAULT SYSDATE,

    ses_id              NUMBER(10),

    rol_id              NUMBER(10) NOT NULL,

    CONSTRAINT pk_permisos
        PRIMARY KEY (perm_id),

    CONSTRAINT fk_permisos_roles
        FOREIGN KEY (rol_id)
        REFERENCES roles (rol_id),

    CONSTRAINT fk_permisos_sesiones
        FOREIGN KEY (ses_id)
        REFERENCES sesiones (ses_id)
);



-- =====================================================
-- TABLA: USUARIOS_ROLES
-- =====================================================

CREATE TABLE usuarios_roles (
    ur_id               NUMBER(10)
        GENERATED BY DEFAULT AS IDENTITY,

    ur_asignacion       DATE DEFAULT SYSDATE,

    ur_inicio           DATE,

    usr_id              NUMBER(10) NOT NULL,

    rol_id              NUMBER(10) NOT NULL,

    ses_id              NUMBER(10),

    CONSTRAINT pk_usuarios_roles
        PRIMARY KEY (ur_id),

    CONSTRAINT fk_usuarios_roles_usuarios
        FOREIGN KEY (usr_id)
        REFERENCES usuarios (usr_id),

    CONSTRAINT fk_usuarios_roles_roles
        FOREIGN KEY (rol_id)
        REFERENCES roles (rol_id),

    CONSTRAINT fk_usuarios_roles_sesiones
        FOREIGN KEY (ses_id)
        REFERENCES sesiones (ses_id)
);

CREATE TABLE profesores (
    id_profesor NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre VARCHAR2(50) NOT NULL,
    apellido VARCHAR2(50) NOT NULL,
    email VARCHAR2(100) UNIQUE NOT NULL
);

CREATE TABLE materias(
    id_materia NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre_materia VARCHAR2(100) NOT NULL
);

CREATE TABLE cursos (
    id_curso VARCHAR2(20) UNIQUE PRIMARY KEY NOT NULL,
    id_profesor NUMBER NOT NULL,
    id_materia NUMBER NOT NULL,
    CONSTRAINT fk_cursos_profesores FOREIGN KEY (id_profesor) REFERENCES profesores(id_profesor),
    CONSTRAINT fk_cursos_materias FOREIGN KEY (id_materia) REFERENCES materias(id_materia)
);



CREATE TABLE alumnos (
    id_alumno NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre VARCHAR2(50) NOT NULL,
    apellido VARCHAR2(50) NOT NULL,
    año_ingreso NUMBER DEFAULT EXTRACT(YEAR FROM SYSDATE) NOT NULL CHECK (año_ingreso >= 2000), 
    email VARCHAR2(100) UNIQUE NOT NULL
);

--alumnos-cursos
CREATE TABLE inscripciones (
    id_inscripcion NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_alumno NUMBER NOT NULL,
    id_curso VARCHAR2(20) NOT NULL,
    CONSTRAINT fk_inscripciones_alumnos FOREIGN KEY (id_alumno) REFERENCES alumnos(id_alumno),
    CONSTRAINT fk_inscripciones_curso FOREIGN KEY (id_curso) REFERENCES cursos(id_curso)
);

CREATE TABLE calificaciones (
    id_calificacion NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_inscripcion NUMBER NOT NULL,
    nota NUMBER(4,2) NOT NULL CHECK(nota >= 0 AND nota <= 10),
    estatus VARCHAR2(10) GENERATED ALWAYS AS (CASE WHEN nota >= 6.0 THEN 'Aprobado' ELSE 'Deficiente' END) VIRTUAL,
    CONSTRAINT fk_calificaciones_inscripciones FOREIGN KEY (id_inscripcion) REFERENCES inscripciones(id_inscripcion)
);

CREATE TABLE asistencia (
    id_asistencia NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_inscripcion NUMBER NOT NULL,
    fecha DATE NOT NULL CHECK(fecha <= SYSDATE),
    estado VARCHAR2(20) NOT NULL,
    CONSTRAINT fk_asistencia_inscripcion FOREIGN KEY (id_inscripcion) REFERENCES inscripciones(id_inscripcion)
    
);
